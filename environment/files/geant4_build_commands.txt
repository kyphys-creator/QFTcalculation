# Commands used to install Geant4 11.4.3 on 2026-10-20 (Mac Studio, Apple M4 Max, 36 GB, macOS 15.5, Apple clang 17.0).
# A record, not an unattended script: run step by step and check each result.  Logs: ~/hep/src/logs/geant4-*.log
# Prerequisites: Homebrew cmake 4.4.4; miniforge (conda 25.7.0); ~/hep/opt/buildenv.sh (PATH without the MESA SDK).

## 1. Sources and data (curl: no com.apple.quarantine attribute); 2.48 GB in all
mkdir -p ~/hep/src/geant4-dl && cd ~/hep/src/geant4-dl
curl -fsSL -o geant4-v11.4.3.tar.gz https://cern.ch/geant4-data/releases/geant4-v11.4.3.tar.gz   # 36.0 MB
#   sha256 f192bd580a6e3a4f606000e81c3f7d7089644de2906a42233d9f223b74b4a944
for f in G4NDL.4.7.1 G4EMLOW.8.8 G4PhotonEvaporation.6.1.2 G4RadioactiveDecay.6.1.2 G4PARTICLEXS.4.2 G4PII.1.3 \
         G4RealSurface.2.2 G4SAIDDATA.2.0 G4ABLA.3.3 G4INCL.1.3 G4ENSDFSTATE.3.0 G4CHANNELING.2.0 G4TENDL.1.4; do
  curl -fsSL -o $f.tar.gz https://cern.ch/geant4-data/datasets/$f.tar.gz
done
#   check each file's MD5 against cmake/Modules/G4DatasetDefinitions.cmake of the source (all 13 agreed)

## 2. Qt 6 for the Qt UI and viewers: a separate conda env (Homebrew's qtbase would upgrade freetype, glib, ...)
~/miniforge3/bin/conda create -y -n g4 -c conda-forge --override-channels qt6-main python=3.12 pip
#   qt6-main 6.12.0, python 3.12.15, libcxx 23.1.3 (its libc++ uses /usr/lib/libc++abi.dylib); 55 MB fetched

## 3. Geant4: multithreaded, TLS model global-dynamic (needed by geant4_pybind), Qt 6 from env g4, no GDML
cd ~/hep/src && tar xzf geant4-dl/geant4-v11.4.3.tar.gz
source ~/hep/opt/buildenv.sh
E=~/miniforge3/envs/g4
mkdir -p geant4-v11.4.3/build && cd geant4-v11.4.3/build
cmake .. -DCMAKE_INSTALL_PREFIX=$HOME/hep/geant4-11.4.3 -DCMAKE_BUILD_TYPE=Release \
  -DCMAKE_C_COMPILER=/usr/bin/clang -DCMAKE_CXX_COMPILER=/usr/bin/clang++ \
  -DGEANT4_BUILD_MULTITHREADED=ON -DGEANT4_BUILD_TLS_MODEL=global-dynamic \
  -DGEANT4_USE_QT=ON -DQT_DIR=$E/lib/cmake/Qt6 -DQt6_DIR=$E/lib/cmake/Qt6 \
  -DQt6OpenGLWidgets_DIR=$E/lib/cmake/Qt6OpenGLWidgets \
  -DCMAKE_INSTALL_RPATH_USE_LINK_PATH=ON \
  -DGEANT4_INSTALL_DATA=OFF -DGEANT4_INSTALL_DATASETS_TENDL=ON -DGEANT4_INSTALL_EXAMPLES=ON
#   QT_DIR, Qt6_DIR and Qt6OpenGLWidgets_DIR are each needed (Geant4 finds QT, Qt6 and Qt6OpenGLWidgets separately);
#   CMAKE_PREFIX_PATH=$E is not used, so that expat comes from the macOS SDK and not from the conda env
make -j12 && make install          # 2 min 45 s; peak memory was not a problem
cd ~/hep/geant4-11.4.3/share/Geant4/data && for f in ~/hep/src/geant4-dl/G4*.tar.gz; do tar xzf $f; done   # 3.0 GB
~/hep/geant4-11.4.3/bin/geant4-config --check-datasets       # 13 x INSTALLED
cp files/geant4env.sh ~/hep/opt/geant4env.sh                 # source it before building or running Geant4 programs

## 4. geant4_pybind 0.1.3 into env g4, against this Geant4 (PyPI sdist, 0.4 MB; pybind11-stubgen 2.5.5, 30 kB)
~/miniforge3/envs/g4/bin/pip install pybind11-stubgen==2.5.5
mkdir -p ~/hep/src/geant4pybind-dl && cd ~/hep/src/geant4pybind-dl
~/miniforge3/envs/g4/bin/pip download --no-deps --no-binary :all: geant4-pybind==0.1.3 -d .
#   sha256 4a4cedca4c2a7d596fc9cdffc0fef69fe68fe1652cf11d13d711a73acdedc9e8
mkdir -p x && tar xzf geant4_pybind-0.1.3.tar.gz -C x
cd x/geant4_pybind-0.1.3 && cp -p source/track/pyG4Track.cc source/track/pyG4Track.cc.orig
patch -p1 < files/geant4_pybind_pyG4Track.cc.patch   # 11.4.3 overloads G4Track::IncrementCurrentStepNumber
cd .. && source ~/hep/opt/buildenv.sh
export PATH="$PATH:$E/bin" GEANT4_DATA_DIR=$HOME/hep/geant4-11.4.3/share/Geant4/data   # no download prompt at import
export CMAKE_GENERATOR="Unix Makefiles" CMAKE_BUILD_PARALLEL_LEVEL=6
export CMAKE_ARGS="-DGeant4_DIR=$HOME/hep/geant4-11.4.3/lib/cmake/Geant4 -DCMAKE_C_COMPILER=/usr/bin/clang \
  -DCMAKE_CXX_COMPILER=/usr/bin/clang++ -DCMAKE_POLICY_VERSION_MINIMUM=3.5 -DCMAKE_PREFIX_PATH=$E/lib/cmake \
  -DPython_EXECUTABLE=$E/bin/python3.12 -DCMAKE_INTERPROCEDURAL_OPTIMIZATION=OFF"
#   Python_EXECUTABLE: pybind11 uses FindPython (setup.py's PYTHON_EXECUTABLE is ignored; it found CLT's Python 3.9)
#   CMAKE_INTERPROCEDURAL_OPTIMIZATION=OFF: with pybind11's default full -flto the final link exhausted the 36 GB
#   and froze the Mac twice; without it the build took about 5 min and free memory stayed above 90 %
$E/bin/pip install -v --no-build-isolation --no-deps --no-cache-dir ./geant4_pybind-0.1.3

## 5. Checks
cd ~/Documents/Claude/QFTcalculation/environment/tests/geant4 && ./run_geant4.sh --qt
